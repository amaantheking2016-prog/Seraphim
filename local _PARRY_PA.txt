local function applyFFlag(name, value)
    name = tostring(name or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if name == "" then return false, "empty name" end
    local ok, err = pcall(function()
        if setfflag then
            setfflag(name, tostring(value))
        elseif setfflags then
            setfflags(name, tostring(value))
        else
            error("executor has no setfflag")
        end
    end)
    if ok then
        local map = loadFFlagsMap()
        map[name] = tostring(value)
        saveFFlags(map)
        getgenv()._SeraphimFFlags = map
    end
    return ok, err
end

local function _hook(remote)
    if _reverted[remote] then return end
    local meta = getrawmetatable(remote)
    if not meta then return end
    if not _original[meta] then
        _original[meta] = true
        setreadonly(meta, false)
        local _old = meta.__index
        meta.__index = function(self, key)
            if (key == "FireServer" and self:IsA("RemoteEvent"))
                or (key == "InvokeServer" and self:IsA("RemoteFunction")) then
                return function(_, ...)
                    local _arguments = { ... }
                    if _is_valid(_arguments) then
                        local fullyReady = _PARRY_PATCH.keyTable ~= nil and _PARRY_PATCH.transformFn ~= nil
                        if fullyReady then

                                _PARRY_PATCH.ready = true

                        end
                    end
                    return _old(self, key)(_, unpack(_arguments))
                end
            end
            return _old(self, key)
        end
        setreadonly(meta, true)
    end
end

function _PARRY_PATCH.fire(curveCFrame, screenPositions, mouseLocation)
    if not _PARRY_PATCH.ready or not _PARRY_PATCH.parryRemote then return false end
    local kt = _PARRY_PATCH.keyTable
    if not kt then return false end
    local keyIndex = kt[1]
    local currentKey = kt[2] and kt[2][keyIndex]
    if not currentKey then
        currentKey = kt[3]
        if type(currentKey) ~= "string" and type(currentKey) ~= "number" then
            if type(kt[1]) == "table" and kt[3] then
                currentKey = kt[1][kt[3]]
            end
        end
    end
    if not currentKey then return false end

    local tok, transformed = pcall(_PARRY_PATCH.transformFn, currentKey, "TIME")
    if not tok or not transformed then
        tok, transformed = pcall(_PARRY_PATCH.transformFn, currentKey)
        if not tok or not transformed then return false end
    end

    local serverTime = workspace:GetServerTimeNow() * 100
    local timeStr = tostring(math.floor(serverTime))
    local tc = {}
    for i = 1, #timeStr do
        local ki = (i - 1) % #transformed + 1
        local kb = string.byte(transformed, ki) or 0
        local tb = (string.byte(timeStr, i) + i) % 256
        tc[i] = string.char(bit32.bxor(tb, kb))
    end
    local token = table.concat(tc)

    local fok = pcall(function()
        _PARRY_PATCH.parryRemote:FireServer(
            _PARRY_PATCH.parryHash,
            currentKey,
            token,
            0.5,
            curveCFrame,
            screenPositions,
            mouseLocation,
            false
        )
    end)
    return fok
end

local function flat(v)
    local f = Vector3.new(v.X, 0, v.Z)
    if f.Magnitude < 1e-4 then
        return Vector3.new(0, 0, -1)
    end
    return f.Unit
end

local function bodyFlatLookRight()
    local root = LocalPlayer.Character and LocalPlayer.Character.PrimaryPart
    if root then
        return flat(root.CFrame.LookVector), flat(root.CFrame.RightVector), root.Position
    end
    local cam = workspace.CurrentCamera
    if not cam then
        return Vector3.new(0, 0, -1), Vector3.new(1, 0, 0), Vector3.zero
    end
    return flat(cam.CFrame.LookVector), flat(cam.CFrame.RightVector), cam.CFrame.Position
end

local function curveCFrame()
    local cam = workspace.CurrentCamera
    if not cam then return CFrame.new() end

    local bodyLook, bodyRight, origin = bodyFlatLookRight()
    local mode = Config.CurveMode

    if mode == "Random" then
        local modes = { "Camera", "Left", "Right", "Back", "High", "Low" }
        mode = modes[math.random(1, #modes)]
    end

    if mode == "Camera" then
        return cam.CFrame
    end

    local aim
    if mode == "Left" then
        aim = origin - bodyRight * 10000
    elseif mode == "Right" then
        aim = origin + bodyRight * 10000
    elseif mode == "Back" then
        aim = origin - bodyLook * 10000
    elseif mode == "High" then
        aim = origin + Vector3.new(0, 12000, 0) + bodyLook * 200
    elseif mode == "Low" then
        aim = origin + Vector3.new(0, -9000, 0) + bodyLook * 400
    else
        return cam.CFrame
    end

    return CFrame.new(origin, aim)
end

local curveState = {
    lerp_radians = 0,
    last_warping = 0,
    curving = 0,
}

local function linear_predict(a, b, t)
    return a + (b - a) * t
end

local function is_curved()
    local ball = getPrimaryBall()
    if not ball then return false end
    if not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then return false end
    local zoomies = ball:FindFirstChild("zoomies")
    if not zoomies then return false end
    local velocity = zoomies.VectorVelocity or Vector3.new()
    local speed = velocity.Magnitude
    if speed == 0 then return false end
    local ball_direction = velocity.Unit
    local direction_vector = LocalPlayer.Character.PrimaryPart.Position - ball.Position
    if direction_vector.Magnitude == 0 then return false end
    local direction = direction_vector.Unit
    local dot = direction:Dot(ball_direction)
    local speed_threshold = math.min(speed / 100, 40)
    local direction_difference = ball_direction - velocity
    local direction_similarity = 0
    if direction_difference.Magnitude > 0 then
        direction_similarity = direction:Dot(direction_difference.Unit)
    end
    local dot_difference = dot - direction_similarity
    local distance = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Magnitude
    local ping = Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    local dot_threshold = 0.5 - (ping / 1000)
    local reach_time = distance / speed - (ping / 1000)
    local ball_distance_threshold = 15 - math.min(distance / 1000, 15) + speed_threshold
    local clamped_dot = math.clamp(dot, -1, 1)
    local radians = math.rad(math.asin(clamped_dot))
    curveState.lerp_radians = linear_predict(curveState.lerp_radians, radians, 0.8)
    if speed > 0 and reach_time > ping / 10 then
        ball_distance_threshold = math.max(ball_distance_threshold - 15, 15)
    end
    if distance < ball_distance_threshold then return false end
    if dot_difference < dot_threshold then return true end
    if curveState.lerp_radians < 0.018 then
        curveState.last_warping = tick()
    end
    if (tick() - curveState.last_warping) < (reach_time / 1.5) then return true end
    if (tick() - curveState.curving) < (reach_time / 1.5) then return true end
    return dot < dot_threshold
end

local function isTargeted(ball)
    local t = ball:GetAttribute("target")
    return t == LocalPlayer.Name or t == LocalPlayer.DisplayName
end

local function getPing()
    local ok, v = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    return (ok and typeof(v) == "number") and v or 60
end

local State = {
    lastParry = 0,
    lockedUntil = 0,
    parried = false,
    conn = nil,
}

local function divisor_mult()
    local acc = math.clamp(Config.Accuracy or 100, 1, 100)
    return 0.7 + (acc - 1) * 0.0035353535353535
end

local function tryParry(ball)
    local now = tick()
    if State.parried or now < State.lockedUntil then return end
    if (now - State.lastParry) < Config.MinCooldown then return end
    if not Config.Enabled or not _PARRY_PATCH.ready then return end
    if not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then return end

    local zoomies = ball:FindFirstChild("zoomies")
    if not zoomies or not isTargeted(ball) then return end

    if Config.AntiCurve and is_curved() then return end

    local velocity = zoomies.VectorVelocity or Vector3.new()
    local speed = velocity.Magnitude
    if speed < 1 then return end

    local distance = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Magnitude
    local ping = getPing() / 10
    local ping_threshold = math.clamp(ping / 10, 5, 17)
    local capped = math.min(math.max(speed - 9.5, 0), 650)
    local speed_divisor = (2.4 + capped * 0.002) * divisor_mult()
    local parry_accuracy = (ping_threshold + math.max(speed / math.max(speed_divisor, 0.4), 9.5)) * Config.RangeMult
    if distance > parry_accuracy then return end

    State.parried = true
    State.lastParry = now
    State.lockedUntil = now + Config.MinCooldown

    fireParry()

    pcall(function()
        ball:GetAttributeChangedSignal("target"):Once(function()
            State.parried = false
            State.lockedUntil = 0
        end)
    end)
    task.delay(math.max(Config.MinCooldown, 0.6), function()
        State.parried = false
    end)
end

local function startAP()
    if State.conn then State.conn:Disconnect() end
    State.conn = RunService.Heartbeat:Connect(function()
        if not Config.Enabled or not _PARRY_PATCH.ready then return end
        if State.parried or tick() < State.lockedUntil then return end
        for _, ball in ipairs(getBalls()) do
            tryParry(ball)
            if State.parried then break end
        end
    end)
end

local function stopAP()
    if State.conn then State.conn:Disconnect() State.conn = nil end
    State.parried = false
    State.lockedUntil = 0
    Config.Enabled = false
end

local ManualSpam = {

local function stopManualSpamLoop()
    ManualSpam.holding = false
    if ManualSpam.conn then
        ManualSpam.conn:Disconnect()
        ManualSpam.conn = nil
    end
end

local function startManualSpamLoop()
    stopManualSpamLoop()
    ManualSpam.holding = true
    local acc = 0
    ManualSpam.conn = RunService.Heartbeat:Connect(function(dt)
        if not ManualSpam.holding or not Config.ManualSpam then
            stopManualSpamLoop()
            return
        end
        if not _PARRY_PATCH.ready then return end
        local cps = math.clamp(Config.SpamCPS or 20, 20, 500)
        acc += (dt or 0.016)
        if acc < (1 / cps) then return end
        acc = 0
        local now = tick()
        if State.parried or now < State.lockedUntil then return end
        if (now - State.lastParry) < 0.05 then return end
        State.parried = true
        State.lastParry = now
        State.lockedUntil = now + 0.08
        fireParry()
        task.delay(0.08, function()
            State.parried = false
        end)
    end)
end

local abilityEspBillboards = {}
local abilityEspConnections = {}
local abilityEspPlayerAddedConnection = nil

local function create_ability_esp_for_player(player)
    task.spawn(function()
        local character = player.Character
        while not character or not character.Parent do
            task.wait()
            character = player.Character
        end

        local head = character:WaitForChild('Head', 10)
        if not head or not getgenv().AbilityESP then
            return
        end

        local existing = head:FindFirstChild('AbilityESPGui')
        if existing then
            existing:Destroy()
        end

        local billboard = Instance.new('BillboardGui')
        billboard.Name = 'AbilityESPGui'
        billboard.Adornee = head
        billboard.Size = UDim2.new(0, 220, 0, 60)
        billboard.StudsOffset = Vector3.new(0, 3.5, 0)
        billboard.AlwaysOnTop = true
        billboard.Parent = head

        local label = Instance.new('TextLabel')
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.TextColor3 = Color3.fromRGB(255, 255, 255)
        label.TextSize = 14
        label.TextStrokeTransparency = 0
        label.Font = Enum.Font.Roboto
        label.RichText = true
        label.TextXAlignment = Enum.TextXAlignment.Center
        label.TextYAlignment = Enum.TextYAlignment.Center
        label.Parent = billboard
        label.Visible = false

        abilityEspBillboards[player] = label

        local humanoid = character:FindFirstChild('Humanoid')
        if humanoid then
            humanoid.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
        end

        local heartbeatConnection
        heartbeatConnection = RunService.Heartbeat:Connect(function()
            if not (character and character.Parent) then
                if heartbeatConnection then
                    heartbeatConnection:Disconnect()
                end
                pcall(function()
                    billboard:Destroy()
                end)
                abilityEspBillboards[player] = nil
                return
            end

            if getgenv().AbilityESP then
                label.Visible = true
                local ability = player:GetAttribute('EquippedAbility')
                if ability then
                    label.Text = '<b>' .. player.DisplayName .. ' [' .. ability .. ']' .. '</b>'
                else
                    label.Text = '<b>' .. player.DisplayName .. '</b>'
                end
            else
                label.Visible = false
            end
        end)

        abilityEspConnections[player] = heartbeatConnection
    end)
end

local function add_ability_esp_player(player)
    if player == LocalPlayer then
        return
    end

    if abilityEspConnections[player] then
        pcall(function()
            abilityEspConnections[player]:Disconnect()
        end)
        abilityEspConnections[player] = nil
    end

    player.CharacterAdded:Connect(function()
        create_ability_esp_for_player(player)
    end)

    if player.Character then
        task.spawn(function()
            create_ability_esp_for_player(player)
        end)
    end
end

local function start_ability_esp()
    getgenv().AbilityESP = true

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            add_ability_esp_player(player)
        end
    end

    if not abilityEspPlayerAddedConnection then
        abilityEspPlayerAddedConnection = Players.PlayerAdded:Connect(function(player)
            if getgenv().AbilityESP then
                add_ability_esp_player(player)
            end
        end)
    end
end

local function stop_ability_esp()
    getgenv().AbilityESP = false

    if abilityEspPlayerAddedConnection then
        pcall(function()
            abilityEspPlayerAddedConnection:Disconnect()
        end)
        abilityEspPlayerAddedConnection = nil
    end

    for _, connection in pairs(abilityEspConnections) do
        pcall(function()
            connection:Disconnect()
        end)
    end
    abilityEspConnections = {}

    for _, label in pairs(abilityEspBillboards) do
        pcall(function()
            if label and label.Parent then
                label.Parent:Destroy()
            end
        end)
    end
    abilityEspBillboards = {}
end

        local function scan(folder)
            if not folder then return end
            for _, b in ipairs(folder:GetChildren()) do

                if z then
                    local v = z.VectorVelocity
                    if v then
                        local s = v.Magnitude
                        if s > maxSpeed then maxSpeed = s end
                    end
                end
            end
        end
        scan(workspace:FindFirstChild("Balls"))
        scan(workspace:FindFirstChild("TrainingBalls"))

        if _velLabel then
            _velLabel.Text = string.format("%.0f", maxSpeed)

            end
        end

            _velPeak = maxSpeed
            if _velPeakLabel then
                _velPeakLabel.Text = string.format("PEAK: %.0f", _velPeak)
            end
        end

local SKIN_LAST_EQUIPPED_CONFIG_KEY = "Skin.LastEquippedSword"
local AUTO_CONFIG_FILE = "Seraphim/auto_config.json"

local function readSeraphimAutoConfig()
    local data = {}
    pcall(function()
        if isfile and isfile(AUTO_CONFIG_FILE) then
            local decoded = HttpService:JSONDecode(readfile(AUTO_CONFIG_FILE))
            if type(decoded) == "table" then
                data = decoded
            end
        end
    end)
    return data
end

local function writeSeraphimAutoConfig(data)
    pcall(function()
        if isfolder and makefolder and not isfolder("Seraphim") then
            makefolder("Seraphim")
        end
        if writefile then
            writefile(AUTO_CONFIG_FILE, HttpService:JSONEncode(data or {}))
        end
    end)
end

local function loadLastEquippedSword()
    local data = readSeraphimAutoConfig()
    local saved = data[SKIN_LAST_EQUIPPED_CONFIG_KEY]
    return type(saved) == "string" and saved or ""
end

getgenv().saveLastEquippedSword = function(swordName)
    if type(swordName) ~= "string" or swordName == "" then return end
    local data = readSeraphimAutoConfig()
    data[SKIN_LAST_EQUIPPED_CONFIG_KEY] = swordName
    writeSeraphimAutoConfig(data)
end

do
    local savedLastSword = loadLastEquippedSword()
    getgenv().skinChanger = getgenv().skinChanger or savedLastSword ~= ""
    getgenv().swordModel = type(getgenv().swordModel) == "string" and getgenv().swordModel ~= "" and getgenv().swordModel or savedLastSword
    getgenv().swordAnimations = type(getgenv().swordAnimations) == "string" and getgenv().swordAnimations ~= "" and getgenv().swordAnimations or savedLastSword
    getgenv().swordFX = type(getgenv().swordFX) == "string" and getgenv().swordFX ~= "" and getgenv().swordFX or savedLastSword
end

task.spawn(function()

    local swordInstancesInstance = rs:WaitForChild("Shared", 9e9):WaitForChild("ReplicatedInstances", 9e9):WaitForChild("Swords", 9e9)
    local swordInstances = require(swordInstancesInstance)

    local swordsController
    task.spawn(function()
        while task.wait(0.25) and not swordsController do
            local ok, conns = pcall(getconnections, rs.Remotes.FireSwordInfo.OnClientEvent)
            if ok and conns then
                for _, v in ipairs(conns) do
                    if v.Function and islclosure and islclosure(v.Function) then
                        local ok2, up = pcall(getupvalues, v.Function)
                        if ok2 and #up == 1 and type(up[1]) == "table" then
                            swordsController = up[1]
                            break
                        end
                    end
                end
            end
        end
    end)

    local function getSlashName(swordName)
        local ok, sln = pcall(function() return swordInstances:GetSword(swordName) end)
        return (ok and sln and sln.SlashName) or "SlashEffect"
    end

    local function refreshSlashName()
        local fxName = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
        if fxName ~= "" then
            getgenv().slashName = getSlashName(fxName)
        else
            getgenv().slashName = "SlashEffect"
        end
    end
    refreshSlashName()

    local function setSword()
        if not getgenv().skinChanger then return end
        if not LocalPlayer.Character then return end
        pcall(function()
            local f = rawget(swordInstances, "EquipSwordTo")
            if type(f) == "function" then
                local ups = getupvalues(f)
                for i = 1, #ups do
                    if type(ups[i]) == "boolean" then
                        setupvalue(f, i, false)
                        break
                    end
                end
            end
        end)
        pcall(function()
            swordInstances:EquipSwordTo(LocalPlayer.Character, getgenv().swordModel)
        end)
        task.spawn(function()
            local attempts = 0
            while not swordsController and attempts < 20 do
                task.wait(0.5); attempts = attempts + 1
            end
            if not swordsController then return end
            pcall(function()
                if swordsController.SetSword then
                    swordsController:SetSword(getgenv().swordAnimations ~= "" and getgenv().swordAnimations or getgenv().swordModel)
                end
            end)
            pcall(function()
                local targetSword = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
                if rs.Remotes:FindFirstChild("FireSwordInfo") then
                    rs.Remotes.FireSwordInfo:FireServer(targetSword)
                end
                if swordsController.currentSword ~= nil then
                    pcall(function() swordsController.currentSword = targetSword end)
                end
                if swordsController.SwordFX ~= nil then
                    pcall(function() swordsController.SwordFX = targetSword end)
                end
            end)
        end)
    end

    local hookedFuncs = {}
    task.spawn(function()
        local remotesToHook = {"ParrySuccessAll", "ParryAttempt", "ParrySuccess", "PlaySound", "PlayVisuals"}
        while task.wait(1) do
            for _, remoteName in ipairs(remotesToHook) do
                local remote = rs.Remotes:FindFirstChild(remoteName)
                if remote and remote:IsA("RemoteEvent") then
                    local ok, conns = pcall(getconnections, remote.OnClientEvent)
                    if ok and type(conns) == "table" then
                        for _, v in ipairs(conns) do
                            local func = v.Function
                            if func and not hookedFuncs[func] then
                                hookedFuncs[func] = true
                                v:Disable()
                                local targetFunc = func
                                local ourFunc
                                ourFunc = function(...)
                                    local args = { ... }
                                    local isLocal = false
                                    for _, arg in ipairs(args) do
                                        if tostring(arg) == LocalPlayer.Name or (typeof(arg) == "Instance" and (arg == LocalPlayer.Character or arg == LocalPlayer)) then
                                            isLocal = true
                                            break
                                        end
                                    end
                                    if isLocal and getgenv().skinChanger then
                                        local fxSword = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
                                        refreshSlashName()
                                        local swordFound = false
                                        local slashFound = false
                                        for i, arg in ipairs(args) do
                                            if type(arg) == "string" then
                                                if fxSword ~= "" and not slashFound and (arg:match("Slash") or arg == "Default" or arg:match("Effect")) then
                                                    args[i] = getgenv().slashName
                                                    slashFound = true
                                                elseif fxSword ~= "" and not swordFound then
                                                    local isSword = false
                                                    pcall(function()
                                                        if rs.Shared.ReplicatedInstances.Swords:FindFirstChild(arg) then
                                                            isSword = true
                                                        end
                                                    end)
                                                    if isSword or arg == LocalPlayer:GetAttribute("CurrentlyEquippedSword") then
                                                        args[i] = fxSword
                                                        swordFound = true
                                                    end
                                                end
                                            end
                                        end
                                        if fxSword ~= "" and not slashFound and type(args[1]) == "string" then
                                            args[1] = getgenv().slashName
                                        end
                                        if fxSword ~= "" and not swordFound and type(args[3]) == "string" then
                                            args[3] = fxSword
                                        end
                                    end
                                    if setthreadidentity then pcall(setthreadidentity, 2) end
                                    pcall(targetFunc, unpack(args))
                                end
                                hookedFuncs[ourFunc] = true
                                remote.OnClientEvent:Connect(ourFunc)
                            end
                        end
                    end
                end
            end
        end
    end)

    getgenv().updateSword = function()

local function setShopButtonText(button, text)
    if not button then return end
    for _, object in ipairs(button:GetDescendants()) do
        if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
            object.Text = text
        end
    end
    if button:IsA("TextButton") then button.Text = text end
end

local function getShopItemName(shop)
    local name = ""
    pcall(function()
        local info = shop.Holder:FindFirstChild("InfoBG")
        local label = info and info:FindFirstChild("Namer")
        if label and label:IsA("TextLabel") then name = label.Text end
    end)
    if name == "" then name = getgenv()._srhCurrentInfoItem or "" end
    return name
end

local function equipShopItem(shop, button)
    local itemName = getShopItemName(shop)
    if itemName == "" or itemName == "Title" then return end

    setShopButtonText(button, "Equipped")

    getgenv().swordModel = itemName
    getgenv().swordAnimations = itemName
    getgenv().swordFX = itemName
    getgenv().skinChanger = true
    if getgenv().updateSword then task.spawn(getgenv().updateSword) end

    task.spawn(function()
        local untilTime = os.clock() + 2.5
        while button.Parent and os.clock() < untilTime do
            setShopButtonText(button, "Equipped")
            task.wait(0.05)
        end
    end)
end

local function unlockSwordShopCards(shop)
    if getgenv()._seraphimSwordShopLoop then return end
    getgenv()._seraphimSwordShopLoop = true
    getgenv().skinChanger = true

    task.spawn(function()
        while shop.Parent do
            local holder = shop:FindFirstChild("Holder")
            local pages = holder and holder:FindFirstChild("Pages")
            local page = pages and pages:FindFirstChild("Sword")

            if page then
                local header = page:FindFirstChild("HeaderTitle")
                if header then header.Visible = false end

                for _, child in ipairs(page:GetDescendants()) do
                    if child.Name == "Lock" and child:IsA("GuiObject") then
                        child.Visible = false
                        local card = child.Parent
                        if card and card:IsA("GuiObject") then
                            card.LayoutOrder = 0
                            if card.Parent and card.Parent.Name == "Unowned" then
                                local owned = page:FindFirstChild("Owned", true)
                                if owned and owned:IsA("GuiObject") then card.Parent = owned end
                            end

                            if not card:FindFirstChild("SeraphimSwordItemHook") then
                                local tag = Instance.new("BoolValue")
                                tag.Name = "SeraphimSwordItemHook"
                                tag.Parent = card

                                local itemName = card.Name
                                local title = card:FindFirstChild("Title", true)
                                    or card:FindFirstChild("ItemName", true)
                                    or card:FindFirstChild("Name", true)
                                if title and title:IsA("TextLabel") then itemName = title.Text end

                                card.InputEnded:Connect(function(input)
                                    if input.UserInputType == Enum.UserInputType.MouseButton1
                                        or input.UserInputType == Enum.UserInputType.Touch then
                                        getgenv()._srhCurrentInfoItem = itemName
                                    end
                                end)
                            end
                        end
                    end
                end
            end

            local info = holder and holder:FindFirstChild("InfoBG")
            local button = info and (info:FindFirstChild("BuyButton") or info:FindFirstChild("EquipButton"))
            if not button and info then
                for _, child in ipairs(info:GetChildren()) do
                    if (child:IsA("TextButton") or child:IsA("ImageButton"))
                        and not child.Name:lower():find("close", 1, true) then
                        button = child
                        break
                    end
                end
            end

            if button then
                button.Visible = true
                if not button:FindFirstChild("SeraphimSwordEquipHook") then
                    local tag = Instance.new("BoolValue")
                    tag.Name = "SeraphimSwordEquipHook"
                    tag.Parent = button
                    button.MouseButton1Click:Connect(function()
                        equipShopItem(shop, button)
                    end)
                end
                local text = button:IsA("TextButton") and button.Text or ""
                if text ~= "Equipped" then setShopButtonText(button, "Equip") end
            end

            task.wait(0.15)
        end
        getgenv()._seraphimSwordShopLoop = false
    end)
end

local function loadUI()
    local url = "https://raw.githubusercontent.com/AchaoticSoftworksCore/AchaoticAssets/main/UiLibrarys/NEVERLOSE-UI-Nightly.luau"
    local src
    if game.HttpGet then
        src = game:HttpGet(url)
    elseif syn and syn.request then
        src = syn.request({ Url = url, Method = "GET" }).Body
    elseif http_request then
        src = http_request({ Url = url, Method = "GET" }).Body
    elseif request then
        src = request({ Url = url, Method = "GET" }).Body
    end
    assert(src and #src > 50, "NEVERLOSE download failed")
    return loadstring(src)()
end

task.spawn(function()
    local NEVERLOSE
    local ok, err = pcall(function() NEVERLOSE = loadUI() end)
    if not ok or not NEVERLOSE then
        warn("[Seraphim] UI failed:", err)
        return
    end
    pcall(function() NEVERLOSE:Theme("Dark") end)
    local Notify = NEVERLOSE:Notification()

    local function n(kind, title, body, t)
        pcall(function() Notify:Notify(kind or "info", title or "Seraphim", body or "", t or 3) end)
    end
    getgenv()._SeraphimNotify = n

    local Window = NEVERLOSE:AddWindow("Seraphim", "Blade Ball", UDim2.new(0.24, 220, 0.32, 180))
    local MainTab = Window:AddTab("Main", "ads")
    local AntiTab = Window:AddTab("Anti", "list")
    local VisualTab = Window:AddTab("Visual", "folder")
    local FlagTab = Window:AddTab("FFlags", "folder")

    local MainSec = MainTab:AddSection("Auto Parry", "left")
    local CurveSec = MainTab:AddSection("Curves", "right")
    local AntiSec = AntiTab:AddSection("Detections", "left")
    local VisualSec = VisualTab:AddSection("Visuals", "left")
    local SwordSec = VisualTab:AddSection("Sword", "right")
    local FlagSec = FlagTab:AddSection("Flags", "left")

    MainSec:AddToggle("Enable Auto Parry", false, function(state)
        local fully = _PARRY_PATCH.ready and _PARRY_PATCH.parryRemote and _PARRY_PATCH.keyTable and _PARRY_PATCH.transformFn
        if state and not fully then
            n("warning", "Not Armed", "Parry once in a real match first", 4)
            Config.Enabled = false
            return
        end
        Config.Enabled = state
        if state then
            startAP()
            n("success", "Auto Parry", "ON", 2)
        else
            stopAP()
            n("info", "Auto Parry", "OFF", 2)
        end
    end)

    MainSec:AddButton("Check Remote", function()
        local ok = _PARRY_PATCH.ready
            and _PARRY_PATCH.parryRemote ~= nil
            and _PARRY_PATCH.keyTable ~= nil
            and _PARRY_PATCH.transformFn ~= nil
            and _PARRY_PATCH.parryHash ~= nil
        if ok then
            n("success", "Armed", "Remote: " .. tostring(_PARRY_PATCH.parryRemote.Name), 3)
        else
            local missing = {}
            if not _PARRY_PATCH.parryRemote then table.insert(missing, "remote") end
            if not _PARRY_PATCH.keyTable then table.insert(missing, "keys") end
            if not _PARRY_PATCH.transformFn then table.insert(missing, "transform") end
            if not _PARRY_PATCH.ready then table.insert(missing, "manual parry") end
            n("warning", "Not Armed", table.concat(missing, ", ") .. " - parry once", 5)
        end
    end)

    MainSec:AddButton("Test Fire", function()
        if not _PARRY_PATCH.ready then
            n("error", "Fail", "Not armed - parry once first", 4)
            return
        end
        local sent = fireParry()
        n(sent and "success" or "error", "Test Fire", sent and "Sent OK" or "Fire failed", 3)
    end)

    CurveSec:AddDropdown("Curve Method", {
        "Camera", "Left", "Right", "Back", "High", "Low", "Random"
    }, "Camera", function(val)
        Config.CurveMode = val
        n("info", "Curve", tostring(val), 2)
    end)

    local SpamSec = MainTab:AddSection("Spam", "left")
    SpamSec:AddToggle("Spam Button", false, function(state)
        Config.ManualSpam = state
        if state then createSpamButton() else destroySpamButton() end
    end)
    SpamSec:AddSlider("Spam CPS", 20, 500, 20, function(v)
        Config.SpamCPS = v
    end)

    AntiSec:AddToggle("Anti Curve", false, function(state)
        Config.AntiCurve = state
        n("info", "Anti Curve", state and "ON" or "OFF", 2)
    end)

    VisualSec:AddToggle("Ability ESP", false, function(state)
        if state then
            start_ability_esp()
            n("success", "Ability ESP", "ON", 2)
        else
            stop_ability_esp()
            n("info", "Ability ESP", "OFF", 2)
        end
    end)

    VisualSec:AddToggle("Ball Velocity", false, function(state)
        getgenv()._seraphimVelocityOn = state
        if state then
            startVelocityOverlay()
            n("success", "Ball Velocity", "ON", 2)
        else
            stopVelocityOverlay()
            n("info", "Ball Velocity", "OFF", 2)
        end
    end)

    SwordSec:AddToggle("Sword Unlock All", false, function(state)
        getgenv().SwordUnlockAll = state
        if state then
            enableSwordUnlockAll()
            n("success", "Sword Unlock All", "ON", 2)
        else
            disableSwordUnlockAll()
            n("info", "Sword Unlock All", "OFF", 2)
        end
    end)

    SwordSec:AddButton("Reapply Sword", function()
        if not getgenv().skinChanger then
            n("warning", "Sword", "Enable Sword Unlock All first", 3)
            return
        end
        if getgenv().updateSword then
            task.spawn(getgenv().updateSword)
            n("success", "Sword", "Reapplied", 2)
        end
    end)

    FlagSec:AddButton("Open FFlag Panel", function()
        openFFlagPanel()
    end)

    FlagSec:AddButton("Rejoin (load FFlags)", function()
        n("info", "Rejoin", "Teleporting...", 2)
        pcall(function()
            TeleportService:Teleport(game.PlaceId, LocalPlayer)
        end)
    end)

    getgenv().SeraphimSetFlag = function(name, value)
        local ok, err = applyFFlag(name, value)
        if ok then
            n("success", "FFlag", tostring(name) .. " = " .. tostring(value), 3)
        else
            n("error", "FFlag Fail", tostring(err), 4)
        end
        return ok
    end
end)

getgenv().Seraphim = {
    Config = Config,
    Patch = _PARRY_PATCH,
    Fire = fireParry,
    Start = startAP,
    Stop = stopAP,
}
